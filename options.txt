################# AUTO SAVE ####################
# Live backups use save hold, save query, and save resume.
# Modern save query lists files as WorldName/db/file:size (relative to worlds/).
# MCBEPlay copies each listed file, then truncates the copy to that size.
#
# BDS-2733 (MANIFEST and .log locked during save hold) was closed in 2022.
# It is not a reason to avoid live backups anymore.
# https://mojira.dev/BDS-2733
#
# auto_save still defaults to false. Set it to true to flush the world every
# save_interval minutes, and to take a live backup when backup_interval has
# elapsed. Both values are minutes.
#
# save_interval is how often the database is saved.
# backup_interval is how often a backup is taken during that save.

# Auto Save System: Saves every x minutes
auto_save=false

# Auto Save Interval (Minutes)
save_interval=60

# Auto Backup Interval (Minutes)
# How often a live backup is taken during a save (minutes)
backup_interval=60

########################################

# Auto Restart System
# Options: time|interval|off
# time: restart at the times in restart_times.
# interval: restart every restart_interval hours. restart_interval=0 disables this.
# off, false, or none: do not schedule restarts.
auto_restart=time

# If you would prefer the server to restart at specific intervals, then set the number of hours between restarts.
# Integer: restart every X hours. 0 disables interval restarts.
restart_interval=12

# If you would prefer the server to restart at specific times of day then use the time option and list the times here
# Comma Separated Times 06:00,12:00,18:00 etc					
restart_times=00:00,06:00,12:00,18:00

# Keep Alive
# If true then the system will restart the server if it's not running. Ideal to prevent it staying off after a crash.
keepalive=false


#### THE ROLL BACK SYSTEM IS OUTDATED AND NOT WORKING FULLY. DO NOT USE! ####
# What should happen if rollback is set to true, is that if the server
# cannot load the world after a number of attempts, it will copy the 
# broken world to the saves folder with the prefix "FAILED_" and then 
# overwrite the server with the most recent successful backup that was taken.
# This is untested as issues were found with the Mojang BDS bug as noted above.

# Rollback
# If server cannot start the system will attempt to rollback to a previous save and try again
# The failed world is backed up in the saves folder prefix FAILED_ before overwriting
# The rolled back save is prefixed with ROLLBACK_ to identify which save was used
rollback=false

################# BDS UPDATES ####################
# Optional updates from Mojang's official download API. Off by default.
# There is no telemetry. A check runs only when this is enabled, or when
# Check for Update is clicked. The only request is the download-link API.
#
# auto_update:
#   false     - do not check automatically (default)
#   interval  - check every update_interval hours
#   time      - check once a day, at or after update_time (HH:MM)
#   true      - same as interval
# A newer Windows stable build warns players, stops the server, copies the
# world into the saves folder, and replaces server binaries and resources.
# The worlds folder, server.properties, allowlist.json, whitelist.json, permissions.json
# and packetlimitconfig.json are kept. On failure the previous install is
# restored and, if it was running, started again.

auto_update=false

# Hours between checks when auto_update=interval (minimum 1)
update_interval=24

# Time of day (HH:MM) when auto_update=time
update_time=04:00
